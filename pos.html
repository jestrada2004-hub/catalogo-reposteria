// --- COTIZAR CON SUBIDA DE PDF A SUPABASE STORAGE (CORREGIDO) --- document.getElementById('btn-cotizacion').addEventListener('click', async () => { if (!bill.length) return; const cliente = document.getElementById('cliente-nombre').value.trim() || 'Consumidor Final'; const telefono = document.getElementById('cliente-telefono').value.trim(); const { sub, desc, total } = updateTotals(); const btn = document.getElementById('btn-cotizacion'); btn.disabled = true; btn.textContent = 'Subiendo PDF...'; const { data, error } = await sb.rpc('guardar_cotizacion_pos', { p_cliente_nombre: cliente, p_cliente_telefono: telefono || null, p_descuento: desc, p_notas: null, p_items: bill }); if (error) { alert('Error al guardar cotización: ' + error.message); btn.disabled = false; btn.textContent = '💬 Cotizar'; return; } let publicUrl = ''; try { const pdfBlob = generarCotizacionBlob(data.id, cliente, telefono, [...bill], sub, desc, total); const fileName = `Cotizacion-${data.id}-${cliente.replace(/[^a-zA-Z0-9]/g, '_')}.pdf`; // Sube el archivo al bucket "facturas" const { error: uploadErr } = await sb.storage.from('facturas').upload(fileName, pdfBlob, { upsert: true, contentType: 'application/pdf' }); if (!uploadErr) { const { data: urlData } = sb.storage.from('facturas').getPublicUrl(fileName); publicUrl = urlData.publicUrl; } } catch (e) { console.log('Error subiendo PDF a Storage:', e); } // Abre WhatsApp con el enlace de descarga incluido const waPhone = formatPhoneHN(telefono); if (waPhone) { const lineas = bill.map(i => `• ${i.cantidad}x ${i.descripcion}: ${money(i.cantidad * i.precio_unitario)}`).join('\n'); let msg = `¡Hola ${cliente}! Adjunto tu *Cotización #${data.id}* de *${BUSINESS_NAME}*:\n\n${lineas}\n\n*Total Estimado: ${money(total)}*`; if (publicUrl) { msg += `\n\n📄 Descarga tu PDF oficial aquí:\n${publicUrl}`; } window.open(`https://wa.me/${waPhone}?text=${encodeURIComponent(msg)}`, '_blank'); } alert(`¡Cotización #${data.id} guardada con éxito!`); bill = []; document.getElementById('cliente-nombre').value = 'Consumidor Final'; document.getElementById('cliente-telefono').value = ''; document.getElementById('input-descuento').value = '0'; renderBill(); btn.disabled = false; btn.textContent = '💬 Cotizar'; });
